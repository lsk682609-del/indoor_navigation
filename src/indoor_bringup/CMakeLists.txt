cmake_minimum_required(VERSION 3.8)
project(indoor_bringup)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
add_executable(livox_cloud_converter scripts/livox_cloud_converter.cpp)
ament_target_dependencies(livox_cloud_converter rclcpp sensor_msgs livox_ros_driver2)
install(TARGETS livox_cloud_converter DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config maps worlds rviz DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/road_network_display.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/velocity_guard.py DESTINATION lib/${PROJECT_NAME})
ament_package()
